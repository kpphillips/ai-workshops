    Runtime sample

    import json, os, re

# Hosted tracing is on by default. Disable it before the SDK opens a trace.
os.environ["OPENAI_AGENTS_DISABLE_TRACING"] = "1"

from agents import Agent, Runner, RunConfig, function_tool, set_tracing_disabled

set_tracing_disabled(True)

@function_tool
def check_coverage(zip_code: str) -> str:
    """Return whether zip_code is PLM-01, PLM-02, or PLM-03."""
    return json.dumps({"covered": zip_code in {"PLM-01", "PLM-02", "PLM-03"}})

@function_tool
def list_slots(service: str, day: str) -> str:
    """List windows for one service and day, each marked open or closed."""

@function_tool
def book_appointment(request_id: str, slot_id: str, caller: str, service: str, zip_code: str, confirmed: bool) -> str:
    """Write one appointment only when confirmed is true. Refuse a duplicate request_id."""

@function_tool
def record_escalation(request_id: str, reason: str) -> str:
    """Record an escalation. Do not create an appointment."""

agent = Agent(
    name="Quoinlatch Desk",
    instructions=open("policy.txt", encoding="utf-8").read(),
    tools=[check_coverage, list_slots, book_appointment, record_escalation],
)

def reply(caller_text: str):
    return Runner.run_sync(
        agent,
        caller_text,
        run_config=RunConfig(tracing_disabled=True),
    )

def redact(text: str) -> str:
    text = re.sub(r"\b\d{3}[-.\s]?\d{3}[-.\s]?\d{4}\b", "[phone]", text)
    text = re.sub(r"\b\d{1,5}\s+[A-Za-z]+(?:\s+[A-Za-z]+){0,2}\s(?:Lane|Street|St|Ave|Road|Rd|Drive|Dr)\b", "[street]", text)
    return text

def write_local(record: dict, path: str) -> None:
    safe = dict(record)
    safe["caller"] = "[caller]"
    safe["text"] = redact(safe.get("text", ""))
    with open(path, "a", encoding="utf-8") as handle:
        handle.write(json.dumps(safe, sort_keys=True) + "\n")

# Does not import agents, Runner, or RunConfig.

def grade(transcript, store, scenario):
    rows = [a for a in store["appointments"] if a["request_id"] == scenario["id"]]
    said = transcript[-1]["text"].lower()
    return {"saved": bool(rows), "claimed_booked": "you're booked" in said}
